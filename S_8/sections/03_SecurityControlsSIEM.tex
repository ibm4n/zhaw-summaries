\section{Security Controls \& SIEM}

\subsection{Fundamental Control Principles}
{
\begin{itemize}[noitemsep]
  \item \textbf{Least Privilege} – minimum necessary access
  \item \textbf{Fail-Safe Defaults} – deny by default
  \item \textbf{Complete Mediation} – every access checked
  \item \textbf{Separation of Privilege} – multiple conditions for access
  \item \textbf{Least Common Mechanism} – minimize shared components
  \item \textbf{Open Design} – transparency over obscurity
  \item \textbf{Psychological Acceptability} – usability of security
  \item \textbf{Goal:} reduce attack surface, enforce secure defaults
\end{itemize}
}

\subsection{SIEM Overview}
{
\textbf{Definition:} \textbf{SIEM} = Security Information \& Event Management

Is generally a software platform that:
\begin{itemize}[noitemsep]
  \item Collects, normalizes, stores, correlates, and analyzes security data
  \item Central component of SOC (Security Operations Center)
  \item Supports detection, alerting, forensic analysis, and compliance
  \item Provides situational awareness across the enterprise
  \item Visual output: dashboards, queries, incident timelines
\end{itemize}
}

\subsection{SIEM Components}
{
\begin{itemize}[noitemsep]
  \item \textbf{Sensors:} Sources generating security-relevant data
    \begin{itemize}[noitemsep]
      \item \textbf{NIDS:} Monitors network traffic (e.g., Snort, Suricata)
      \item \textbf{HIDS:} Monitors host-level activity (e.g., OSSEC)
    \end{itemize}

  \item \textbf{Log Collection \& Normalization:}
    \begin{itemize}[noitemsep]
      \item Collect logs from diverse systems (e.g., firewalls, servers)
      \item Normalize to a standard format (timestamp, IPs, type)
    \end{itemize}

  \item \textbf{Asset Inventory:}
    \begin{itemize}[noitemsep]
      \item Maintains list of assets with roles, owners, criticality
      \item Enhances alert context, reduces false positives
    \end{itemize}

  \item \textbf{Vulnerability Scanner:}
    \begin{itemize}[noitemsep]
      \item Scans systems for known CVEs (e.g., Nessus, OpenVAS)
      \item Supports alert prioritization based on actual risk
    \end{itemize}

  \item \textbf{Correlation Engine:}
    \begin{itemize}[noitemsep]
      \item Links events into meaningful patterns
      \item Detects attacker behavior (e.g., brute force, lateral movement)
      \item Enables TTP-based detection
    \end{itemize}
\end{itemize}
}

\subsection{Pyramid of Pain}
{
\begin{itemize}[noitemsep]
  \item Defense model: higher levels = harder for attacker to adapt
  \item Indicators (bottom to top): Hashes, IPs, Domain Names, Network/Host Artifacts, Tools, TTPs
  \item Low-level IOCs (e.g., IPs, hashes) are easy to change, offer limited disruption
  \item TTPs = Tactics, Techniques, Procedures → deeply tied to attacker behavior
  \item Detecting TTPs disrupts entire campaigns, forces costly retooling
  \item Goal: detect \& disrupt attacker TTPs, not just superficial IOCs
\end{itemize}
}
